#!/bin/bash
set -e

# 1. StarCraft II na ext4

cd ~
wget -O SC2.4.10.zip http://blzdistsc2-a.akamaihd.net/Linux/SC2.4.10.zip

# rozpakuj z hasłem, bez pytań

unzip -q -P iagreetotheeula SC2.4.10.zip -d $HOME
rm SC2.4.10.zip

export SC2PATH=$HOME/StarCraftII
grep -qxF 'export SC2PATH=$HOME/StarCraftII' ~/.bashrc || echo 'export SC2PATH=$HOME/StarCraftII' >> ~/.bashrc

# 2. Mapy SMACv2 — od razu do właściwego folderu

mkdir -p $HOME/StarCraftII/Maps/SMAC_Maps

cd /tmp
wget -q https://github.com/oxwhirl/smacv2/releases/download/maps/SMAC_Maps.zip
unzip -q SMAC_Maps.zip -d $HOME/StarCraftII/Maps/SMAC_Maps/
rm SMAC_Maps.zip

ray stop -f
ray start --head \
--port=6379 \
--dashboard-port=8265 \
--ray-client-server-port=10001 \
--node-manager-por

t=8077 \
--object-manager-port=8076 \
--min-worker-port=11000 \
--max-worker-port=11050

$wslIp = (wsl.exe hostname -I).Trim().Split()[0]
$ports = @(6379,8265,10001,8076,8077) + (11000..11050)

foreach ($p in $ports) {
netsh interface portproxy delete v4tov4 listenport=$p listenaddress=0.0.0.0 2>$null
netsh interface portproxy add v4tov4 listenport=$p listenaddress=0.0.0.0 connectport=$p connectaddress=$wslIp
}

New-NetFirewallRule -DisplayName "RayWSL2" -Direction Inbound -Action Allow -Protocol TCP -LocalPort 6379,8265,10001,8076,8077,11000-11050
